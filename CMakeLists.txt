cmake_minimum_required(VERSION 3.20)
project(gamepad_input_bridge VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(GAMEPAD_BUILD_EXAMPLE "Build the selected controller's printing example" ON)
option(GAMEPAD_BUILD_TESTS "Build deterministic and threaded tests" ON)
option(GAMEPAD_STATIC_RUNTIME "Link MinGW runtime statically in executables" ON)
option(GAMEPAD_SANITIZERS "Enable address/undefined sanitizers on GCC/Clang" OFF)
set(GAMEPAD_DEVICE "flydigi_desert_fox" CACHE STRING "Controller profile for the example executable")
set_property(CACHE GAMEPAD_DEVICE PROPERTY STRINGS flydigi_desert_fox generic_xinput)

find_package(Threads REQUIRED)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

add_library(gamepad_input_bridge STATIC src/gamepad.cpp)
add_library(gamepad::input ALIAS gamepad_input_bridge)
set_target_properties(gamepad_input_bridge PROPERTIES EXPORT_NAME input)
target_compile_features(gamepad_input_bridge PUBLIC cxx_std_17)
target_include_directories(gamepad_input_bridge
    PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
           $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE src)
target_link_libraries(gamepad_input_bridge PUBLIC Threads::Threads)
if(WIN32)
    target_sources(gamepad_input_bridge PRIVATE src/backends/xinput_backend.cpp src/platform/windows_waiter.cpp)
    target_compile_definitions(gamepad_input_bridge PRIVATE _WIN32_WINNT=0x0602)
    if(MINGW)
        # MinGW's generic libxinput targets legacy XINPUT1_3.dll. Use the OS DLL.
        target_link_libraries(gamepad_input_bridge PRIVATE xinput1_4)
    else()
        target_link_libraries(gamepad_input_bridge PRIVATE xinput)
    endif()
else()
    target_sources(gamepad_input_bridge PRIVATE src/backends/unavailable_backend.cpp src/platform/portable_waiter.cpp)
    if(GAMEPAD_BUILD_EXAMPLE)
        message(FATAL_ERROR "Native backend currently supports Windows only. For portable core tests set GAMEPAD_BUILD_EXAMPLE=OFF.")
    endif()
endif()

function(gamepad_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
    if(GAMEPAD_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
endfunction()
function(gamepad_executable target)
    gamepad_warnings(${target})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    if(MINGW AND GAMEPAD_STATIC_RUNTIME AND NOT GAMEPAD_SANITIZERS)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endfunction()
gamepad_warnings(gamepad_input_bridge)

if(GAMEPAD_BUILD_EXAMPLE)
    if(NOT GAMEPAD_DEVICE MATCHES "^(flydigi_desert_fox|generic_xinput)$")
        message(FATAL_ERROR "Unknown GAMEPAD_DEVICE: ${GAMEPAD_DEVICE}")
    endif()
    set(example_target "gamepad_${GAMEPAD_DEVICE}")
    add_executable(${example_target} examples/main.cpp "src/devices/${GAMEPAD_DEVICE}.cpp")
    target_include_directories(${example_target} PRIVATE src)
    target_link_libraries(${example_target} PRIVATE gamepad::input)
    gamepad_executable(${example_target})
    install(TARGETS ${example_target} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

if(GAMEPAD_BUILD_TESTS)
    enable_testing()
    add_executable(gamepad_tests tests/tests.cpp)
    target_include_directories(gamepad_tests PRIVATE src)
    target_link_libraries(gamepad_tests PRIVATE gamepad::input)
    gamepad_executable(gamepad_tests)
    add_test(NAME gamepad_tests COMMAND gamepad_tests)
    set_tests_properties(gamepad_tests PROPERTIES TIMEOUT 30)
    add_executable(gamepad_benchmark tests/benchmark.cpp)
    target_include_directories(gamepad_benchmark PRIVATE src)
    target_link_libraries(gamepad_benchmark PRIVATE gamepad::input)
    gamepad_executable(gamepad_benchmark)
endif()

install(TARGETS gamepad_input_bridge EXPORT GamepadTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/GamepadInputBridgeConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/GamepadInputBridgeConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GamepadInputBridge")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/GamepadInputBridgeConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT GamepadTargets NAMESPACE gamepad::
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GamepadInputBridge")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GamepadInputBridgeConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/GamepadInputBridgeConfigVersion.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GamepadInputBridge")
